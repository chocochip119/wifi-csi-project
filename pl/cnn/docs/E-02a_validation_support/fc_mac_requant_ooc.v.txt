`timescale 1ns / 1ps
// Synthesis-only MAC + requant fixture. Memories remain external ports.
// mac_done is the MAC completion pulse, NOT final fc_done (F-03c).
module fc_mac_requant_ooc (
 input wire clk,input wire rst_n,input wire mac_start,input wire [1:0] mac_sel,
 output wire mac_done,
 output wire [8:0] flat_raddr,input wire [63:0] flat_rdata,
 output wire fifo_re,input wire [63:0] fifo_rdata,input wire fifo_empty,
 output wire [4:0] hidden_raddr,input wire [63:0] hidden_rdata,
 output wire [11:0] fcw_raddr,input wire [63:0] fcw_rdata,
 output wire [8:0] fc_param_raddr,input wire [95:0] fc_param_rdata,
 output wire [7:0] rq,output wire rq_valid,output wire [8:0] rq_tag
);
 wire signed [31:0] acc;
 wire acc_valid;
 wire [8:0] acc_tag;
 fc_mac u_mac (
  .clk(clk),.rst_n(rst_n),.mac_start(mac_start),.mac_sel(mac_sel),.mac_done(mac_done),
  .flat_raddr(flat_raddr),.flat_rdata(flat_rdata),.fifo_re(fifo_re),.fifo_rdata(fifo_rdata),.fifo_empty(fifo_empty),
  .hidden_raddr(hidden_raddr),.hidden_rdata(hidden_rdata),.fcw_raddr(fcw_raddr),.fcw_rdata(fcw_rdata),
  .acc(acc),.acc_valid(acc_valid),.acc_tag(acc_tag)
 );
 requant_stage #(.TAG_WIDTH(9),.FC_MODE(1),.PARAM_BASE0(9'd48),.PARAM_BASE1(9'd176),.PARAM_BASE2(9'd304)) u_rq (
  .clk(clk),.rst_n(rst_n),.acc(acc),.acc_valid(acc_valid),.acc_tag(acc_tag),
  .enc_param_raddr(fc_param_raddr),.enc_param_rdata(fc_param_rdata),.rq(rq),.rq_valid(rq_valid),.rq_tag(rq_tag)
 );
endmodule
